1. אֲנִי רוֹצָה מַיִם קָרִים.
2. אֶתְמוֹל רָצִיתִי קָפֶה.
3. אֶתְמוֹל שָׁתִיתִי מִיץ.
4. אֲנִי אוֹכֶלֶת לֶחֶם וּגְבִינָה.
5. אֶתְמוֹל הָלַכְתִּי לַעֲבוֹדָה.
6. הִיא כּוֹתֶבֶת מִכְתָּב לְאִמָּא.
7. כָּתַבְתִּי הוֹדָעָה קְצָרָה.
8. אֲנִי קוֹרֵאת סֵפֶר בָּעֶרֶב.
9. אֲנַחְנוּ לוֹמְדִים עִבְרִית בָּאוּלְפָּן.
10. לָמַדְתִּי מִלִּים חֲדָשׁוֹת.
11. דָּנִיאֵל גָּר בִּירוּשָׁלַיִם.
12. גַּרְתִּי בְּחֵיפָה שָׁנָה.
13. הִיא קוֹנָה פֵּרוֹת בַּשּׁוּק.
14. אֶתְמוֹל קָנִיתִי חָלָב.
15. עָבַדְתִּי כָּל הַיּוֹם.
16. אַתְּ יוֹדַעַת אֶת הַתְּשׁוּבָה?
17. לֹא יָדַעְתִּי אֶת זֶה.
18. אֲנִי רוֹאָה אֶת הַיָּם.
19. רָאִיתִי אֶת הַחֲבֵר בָּרְחוֹב.
20. הִיא בָּאָה לַבַּיִת עַכְשָׁיו.
21. מָה אַתְּ עוֹשָׂה הַיּוֹם?
22. עָשִׂיתִי שִׁעוּרֵי בַּיִת.
23. אֲנִי שׁוֹמַעַת מוּזִיקָה.
24. שָׁמַעְתִּי אֶת הַשִּׁיר הַזֶּה.
25. פָּגַשְׁתִּי אוֹתוֹ בַּבֹּקֶר.
26. פָּתַחְתִּי אֶת הַחַלּוֹן.
27. סָגַרְתִּי אֶת הַדֶּלֶת.
28. אֲנַחְנוּ יוֹשְׁבִים בְּבֵית קָפֶה.
29. יָשַׁבְתִּי עַל הַכִּסֵּא.
30. עָמַדְתִּי בַּתּוֹר בַּסּוּפֶּר.
31. אֲנִי לוֹקַחַת תַּפּוּחַ וּמַיִם.
32. לָקַחְתִּי אֶת הַמַּפְתֵּחַ.
33. הִיא נוֹתֶנֶת מַתָּנָה לַחֲבֵר.
34. נָתַתִּי לוֹ אֶת הַסֵּפֶר.
35. הַתַּלְמִיד שׁוֹאֵל אֶת הַמּוֹרֶה.
36. שָׁאַלְתִּי שְׁאֵלָה פְּשׁוּטָה.
37. הַמּוֹרֶה עוֹנֶה לַתַּלְמִיד.
38. עָנִיתִי עַל הַשְּׁאֵלָה.
39. אָמַרְתִּי לוֹ תּוֹדָה רַבָּה.
40. אֶתְמוֹל נָסַעְתִּי בָּרַכֶּבֶת לְחֵיפָה.
41. אֲנִי חוֹזֶרֶת הַבַּיְתָה בָּעֶרֶב.
42. דָּנִי רָץ בַּפַּארְק בַּבֹּקֶר.
43. אֶתְמוֹל רַצְתִּי חֲמִשָּׁה קִילוֹמֶטְרִים.
44. אֲנִי עוֹזֶרֶת לַחֲבֵר בָּאוּלְפָּן.
45. שָׂרָה עָזְרָה לְאִמָּא בַּבַּיִת.
46. אֲנִי זוֹכֶרֶת אֶת כָּל הַמִּלִּים.
47. זָכַרְתִּי אֶת מִסְפַּר הַטֶּלֶפוֹן.
48. שָׁכַחְתִּי אֶת הַתִּיק בַּבַּיִת.
49. אֲנִי חוֹשֶׁבֶת עַל הָעֲבוֹדָה.
50. אֶתְמוֹל חָשַׁבְתִּי עַל הָרַעְיוֹן.
51. מָצָאתִי עֲבוֹדָה טוֹבָה בָּעִיר.
52. אֲנִי אוֹהֶבֶת קָפֶה חָזָק בַּבֹּקֶר.
53. אָהַבְתִּי אֶת הַסֶּפֶר הַזֶּה מְאֹד.
54. שַׂמְתִּי אֶת הַכֶּסֶף בַּתִּיק.
55. אֲנִי קָמָה מוּקְדָּם בַּבֹּקֶר תָּמִיד.
56. הַיּוֹם קַמְתִּי בְּשָׁעָה שֶׁבַע.
57. אֶתְמוֹל יָשַׁנְתִּי שְׁמוֹנֶה שָׁעוֹת טוֹבוֹת.
58. שַׂבָּא נָח בַּבַּיִת בַּצָּהֳרַיִם.
59. בַּשַּׁבָּת נַחְנוּ כָּל הַיּוֹם.
60. אֲנַחְנוּ טָסִים מָחָר לְאֵילַת.
61. בַּשָּׁנָה שֶׁעָבְרָה טַסְתִּי לְפָרִיז.
62. הַסּוֹחֵר מוֹכֵר פֵּרוֹת בַּשּׁוּק.
63. אֲנִי בּוֹחֶרֶת יְרָקוֹת טְרִיִּים בַּחֲנוּת.
64. בָּחַרְתִּי מָנָה טְעִימָה בַּמִּסְעָדָה.
65. הָרוֹפֵא בּוֹדֵק אֶת הַחוֹלֶה הַיּוֹם.
66. בָּדַקְתִּי אֶת הַחֶשְׁבּוֹן בַּקְּפֵדָה.
67. שָׁמַרְתִּי לְךָ מָקוֹם טוֹב.
68. אֶתְמוֹל עָבַרְנוּ אֶת הַמִּבְחָן בְּהַצְלָחָה.
69. הָאוֹטוֹבּוּס עוֹצֵר בַּתַּחֲנָה הַקְּרוֹבָה.
70. הַנַּהָג עָצַר בָּאוֹר הָאָדֹם.
71. אֲנִי מְדַבֶּרֶת עִם הַמּוֹרֶה עִבְרִית.
72. אֶתְמוֹל דִּבַּרְתִּי עִם דָּוִד בַּטֶּלֶפוֹן.
73. בִּקַּשְׁתִּי חֶשְׁבּוֹן בַּמִּסְעָדָה בְּנִימוּס.
74. שִׁילַּמְתִּי עַל הַקָּפֶה בַּבֹּקֶר.
75. סַבָּא מְסַפֵּר סִפּוּר יָפֶה לַיְּלָדִים.
76. סִפַּרְתִּי לַחֲבֵר עַל הַטִּיּוּל.
77. אֲנַחְנוּ מְטַיְּלִים בִּירוּשָׁלַיִם הַיּוֹם בְּיַחַד.
78. בַּשַּׁבָּת טִיַּלְנוּ בַּפַּארְק הַלְּאוּמִּי.
79. אֲנִי מְחַפֶּשֶׂת דִּירָה בְּתֵל-אָבִיב.
80. חִפַּשְׂתִּי אֶת הַמַּפְתֵּחוֹת שָׁעָה שְׁלֵמָה.
81. הִיא מְסַדֶּרֶת אֶת הַחֶדֶר יָפֶה.
82. אֶתְמוֹל סִדַּרְתִּי אֶת הָאֲרוֹן בַּבַּיִת.
83. קִבַּלְתִּי הוֹדָעָה חֲשׁוּבָה מֵהַבַּנְק.
84. הַמּוֹרֶה מְלַמֵּד עִבְרִית בָּאוּלְפָּן.
85. שָׁנָה שְׁלֵמָה לִמַּדְתִּי בְּבֵית-סֵפֶר.
86. הַיְּלָדִים מְשַׂחֲקִים בַּכַּדּוּר בַּחֲצַר.
87. אֶתְמוֹל שִׂחַקְנוּ שַׁחְמָט בַּבַּיִת.
88. אִמָּא בִּשְּׁלָה אֲרוּחַת עֶרֶב נִפְלָאָה.
89. אֶתְמוֹל נִקִּיתִי אֶת כָּל הַדִּירָה.
90. הַיַּלְדָּה מְצַיֶּרֶת פְּרָחִים בְּצִבְעוֹנִיּוּת.
91. צִיַּרְתִּי תְּמוּנָה יָפָה עַל נְיָר.
92. אֲנִי מְצַלֶּמֶת אֶת הַנּוֹף הַיָּפֶה.
93. אֶתְמוֹל צִלַּמְתִּי הַרְבֵּה תְּמוּנוֹת בַּיָּם.
94. אֲנַחְנוּ מְבַקְּרִים אֵצֶל סַבָּא בָּעֶרֶב.
95. בַּשַּׁבָּת בִּקַּרְתִּי בַּמּוּזֵאוֹן בִּירוּשָׁלַיִם.
96. אָסוּר לְעַשֵּׁן בַּמָּקוֹם הַצִּבּוּרִי הַזֶּה.
97. הַטֶּכְנַאי מְתַקֵּן אֶת הַמַּחְשֵׁב הַיּוֹם.
98. תִּקַּנְתִּי אֶת הַטָּעוּת בַּמִּכְתָּב.
99. אַבָּא מְבָרֵךְ עַל הַיַּיִן בְּשַׁבָּת.
100. בֵּרַכְתִּי אֶת הֶחָבֵר לְיוֹם-הוּלֶּדֶת.
101. אֲנִי מְחַכָּה לָאוֹטוֹבּוּס בַּתַּחֲנָה הַמֶּרְכָּזִית.
102. חִכִּיתִי לְךָ עֶשְׂרִים דַּקּוֹת שְׁלֵמוֹת.
103. קִוִּיתִי לִפְגּוֹשׁ אוֹתְךָ הַיּוֹם כָּאן.
104. שִׁנִּיתִי אֶת הַסִּסְמָה שֶׁלִּי בַּמַּחְשֵׁב.
105. אֲנִי מְנַסָּה לְדַבֵּר עִבְרִית.
106. אֶתְמוֹל נִסִּיתִי מָנָה חֲדָשָׁה בַּמִּסְעָדָה.
107. אֲנִי מְאַחֶלֶת לְךָ הַצְלָחָה רַבָּה.
108. אִחַלְנוּ חַג שָׂמֵחַ לְכָל הַמִּשְׁפָּחָה.
109. הִיא מְסַיֶּמֶת אֶת הָעֲבוֹדָה בַּזְּמַן.
110. סִיַּמְתִּי אֶת הַקּוּרְס בְּהַצְלָחָה גְּדוֹלָה.
111. אִרְגַּנְתִּי אֶת הַמִּסְמָכִים שֶׁלִּי הַיּוֹם.
112. אֲנִי מַרְגִּישָׁה מְצוּיָּן הַיּוֹם בַּבֹּקֶר.
113. אֶתְמוֹל הִרְגַּשְׁתִּי עָיֵף מְאֹד בָּעֶרֶב.
114. אֲנִי מַזְמִינָה פִּיצָה חַמָּה לַבַּיִת.
115. הִזְמַנְתִּי מָקוֹם טוֹב בַּמִּסְעָדָה מֵרֹאשׁ.
116. הַשִּׁעוּר שֶׁלָּנוּ מַתְחִיל בְּשָׁעָה תֵּשַׁע.
117. הִתְחַלְתִּי לַעֲבוֹד בַּמִּשְׂרָד הֶחָדָשׁ.
118. הָרַכֶּבֶת מַגִּיעָה בִּזְמַן לַתַּחֲנָה עַכְשָׁיו.
119. הִגַּעְתִּי לָאוּלְפָּן בַּבֹּקֶר בְּשָׁלוֹם.
120. אֲנִי מְבַיֶּנֶת אֶת הַשְּׁאֵלָה הֵיטֵב.
121. לֹא הֵבַנְתִּי אֶת הַמִּלָּה הַזֹּאת.
122. אֲנִי מַכִּירָה אֶת הָאִישׁ הַזֶּה.
123. הִכַּרְתִּי אוֹתוֹ בָּאוּלְפָּן בַּקַּיִץ.
124. הַמּוֹרֶה מַסְבִּיר אֶת הַכְּלָל בְּבֵרוּר.
125. הִסְבַּרְתִּי לַחֲבֵר אֶת הַדֶּרֶךְ.
126. הַתַּלְמִיד מַקְשִׁיב לַמּוֹרֶה בַּכִּתָּה בְּקֶשֶׁב.
127. הִקְשַׁבְתִּי לַחֲדָשׁוֹת בָּרַדְיוֹ בַּבֹּקֶר.
128. אֲנַחְנוּ מַמְשִׁיכִים לִלְמוֹד עִבְרִית בְּשִׂמְחָה.
129. הִמְשַׁכְתִּי יָשָׁר עַד הָרַמְזוֹר.
130. הַמְּנַהֵל מַחְלִיט עַל הַתּוֹכְנִית הַיּוֹם.
131. הֶחְלַטְתִּי לִנְסוֹעַ לִירוּשָׁלַיִם מָחָר.
132. הֵבֵאתִי מַתָּנָה יָפָה מִפָּרִיז.
133. אֲנִי מַדְלִיקָה אֶת הָאוֹר בַּחֶדֶר.
134. אֶתְמוֹל הִדְלַקְתִּי אֶת הַמַּזְגָן בַּבַּיִת.
135. הִיא מְכַבָּה אֶת הָאוֹר בַּלַּיְלָה.
136. כִּבִּיתִי אֶת הַטֶּלֶוִיזְיָה לִפְנֵי הַשֵּׁנָה.
137. הַגֶּשֶׁם מַפְסִיק לָרֶדֶת בַּחוּץ עַכְשָׁיו.
138. הִפְסַקְתִּי לַעֲבוֹד מְאוּחָר בַּלַּיְלָה.
139. הַמֶּלְצַר מַמְלִיץ עַל הַדָּג הַטָּרִי.
140. הִמְלַצְתִּי עַל הַמָּלוֹן הַזֶּה לַחֲבֵרִים.
141. הַתַּלְמִיד מַצְלִיחַ בַּלִּמּוּדִים בָּאוּלְפָּן יָפֶה.
142. הִצְלַחְתִּי בַּמִּבְחָן הַקָּשֶׁה בְּעִבְרִית.
143. אֲנִי מַעֲדִיפָה תֵּה עַל קָפֶה.
144. הֶעֱדַפְתִּי לְהִשָּׁאֵר בַּבַּיִת בַּגֶּשֶׁם.
145. אֲנִי מַסְכִּימָה עִם כָּל מִלָּה.
146. הִסְכַּמְנוּ עַל הַמְּחִיר מִיָּד שָׁם.
147. הִתְרַחַצְתִּי בַּיָּם עִם כָּל הַחֲבֵרִים.
148. הִתְלַבַּשְׁתִּי מַהֵר בַּבֹּקֶר לָעֲבוֹדָה.
149. אֲנִי מִתְרַגֶּשֶׁת מֵהַפְּגִישָׁה הַחֲדָשָׁה הַיּוֹם.
150. מְאֹד הִתְרַגַּשְׁתִּי בַּחֲתוּנָּה שֶׁל אָחִי.
151. אֲנִי מִשְׁתַּמֶּשֶׁת בַּמַּחְשֵׁב בָּעֲבוֹדָה תָּמִיד.
152. הִשְׁתַּמַּשְׁתִּי בַּמִּלּוֹן בַּשִּׁעוּר אֶתְמוֹל.
153. הֵם מִתְחַתְּנִים בַּקַּיִץ הַקָּרוֹב בְּיִשְׂרָאֵל.
154. הָאִישׁ מִתְפַּלֵּל בְּבֵית-הַכְּנֶסֶת בַּבֹּקֶר.
155. הִתְפַּלַּלְנוּ שָׁם בְּיוֹם כִּפּוּר בְּיַחַד.
156. אֲנִי מִצְטַעֶרֶת עַל הַטָּעוּת הַזֹּאת.
157. מְאֹד הִצְטַעַרְתִּי עַל מָה שֶׁקָּרָה.
158. אֲנִי מְתַקְשֶׁרֶת לְאִמָּא בָּעֶרֶב.
159. אֶתְמוֹל הִתְקַשַּׁרְתִּי לָרוֹפֵא בַּבֹּקֶר.
160. אֲנַחְנוּ מִתְרָאִים שׁוּב בָּעֶרֶב בַּקָּפֶה.
161. הִתְרָאֵינוּ בַּקָּפֶה בַּשָּׁבוּעַ שֶׁעָבַר.
162. הִיא מִסְתַּכֶּלֶת עַל הַתְּמוּנָה הַיָּפָה.
163. הִסְתַּכַּלְתִּי בַּחַלּוֹן שָׁעָה אֲרוּכָּה הַיּוֹם.
164. הַתַּלְמִיד מִתְקַדֵּם יָפֶה בָּאוּלְפָּן הַשָּׁנָה.
165. הִתְקַדַּמְתִּי הַרְבֵּה בַּחֹדֶשׁ הָאַחֲרוֹן בָּעֲבוֹדָה.
166. אֲנִי מִתְרַגֶּלֶת לַמֶּזֶג הָאֲוִיר בָּאָרֶץ.
167. הִתְרַגַּלְתִּי מַהֵר לַחַיִּים בְּיִשְׂרָאֵל.
168. הַתַּלְמִיד נִכְנָס לַכִּתָּה בְּשֶׁקֶט.
169. אֶתְמוֹל נִכְנַסְתִּי לַמִּשְׂרָד מוּקְדָּם בַּבֹּקֶר.
170. אֲנִי נִשְׁאֶרֶת בַּבַּיִת הַיּוֹם בָּעֶרֶב.
171. אֶתְמוֹל נִשְׁאַרְתִּי בָּעֲבוֹדָה עַד מְאוּחָר.
172. אֲנַחְנוּ נִפְגָּשִׁים בַּקָּפֶה בָּעֶרֶב בְּיַחַד.
173. אֶתְמוֹל נִפְגַּשְׁתִּי עִם הַמְּנַהֵל לְשִׂיחָה.
174. הַחֲנוּת נִסְגֶּרֶת בְּשָׁעָה שְׁמוֹנֶה בָּעֶרֶב.
175. הַדֶּלֶת נִסְגְּרָה לְפֶתַע פִּתְאוֹם בָּרוּחַ.
176. הַסִּפְרִיָּה נִפְתַּחַת בַּבֹּקֶר בְּשָׁעָה שְׁמוֹנֶה.
177. הַקּוּרְס הֶחָדָשׁ נִפְתַּח אֶתְמוֹל בָּאוּלְפָּן.
178. נִזְהַרְתִּי מְאֹד בַּדֶּרֶךְ הַחֲלָקָה בַּגֶּשֶׁם.
179. תִּינוֹק חָדָשׁ נוֹלָד הַיּוֹם בַּמִּשְׁפָּחָה.
180. נוֹלַדְתִּי בְּיִשְׂרָאֵל לִפְנֵי עֶשְׂרִים שָׁנָה.
181. אֲנִי נִזְכֶּרֶת בַּסִּפּוּר הַיָּפֶה עַכְשָׁיו.
182. פִּתְאוֹם נִזְכַּרְתִּי בַּשֵּׁם שֶׁלּוֹ מִיָּד.
183. קָשֶׁה לְהִפָּרֵד מֵחֲבֵרִים טוֹבִים תָּמִיד.
184. אֶתְמוֹל נִפְרַדְנוּ בִּשְׂדֵה הַתְּעוּפָה בְּחִבּוּק.
185. הַשֶּׁמֶשׁ נֶעֱלֶמֶת מֵאֲחוֹרֵי הֶעָנָן לְאַט.
186. הַמַּפְתֵּחַ שֶׁלִּי נֶעֱלַם בַּבֹּקֶר פִּתְאוֹם.
187. הַתִּינוֹק נִרְדָּם בָּעֶרֶשׂ בְּשֶׁקֶט עַכְשָׁיו.
188. נִרְדַּמְתִּי מַהֵר מוּל הַטֶּלֶוִיזְיָה אֶתְמוֹל.
189. הַסֶּרֶט נִמְשָׁךְ שְׁעָתַיִם תְּמִימוֹת בַּקּוֹלְנוֹעַ.
190. הַיְּשִׁיבָה נִמְשְׁכָה הַרְבֵּה זְמַן אֶתְמוֹל.
191. אַף אֶחָד לֹא נִכְשָׁל כָּאן.
192. נִכְשַׁלְתִּי בַּטֶּסְט הָרִאשׁוֹן שֶׁל נְהִיגָה.
193. אֲנִי מַסְפִּיקָה לָרַכֶּבֶת בַּבֹּקֶר תָּמִיד.
194. הִסְפַּקְתִּי לְסַיֵּם אֶת הַדּוּ״חַ הַיּוֹם.
195. הַמַּצָּב הַכַּלְכָּלִי מַשְׁפִּיעַ עַל הַכֹּל.
196. הַמּוֹרֶה הִשְׁפִּיעַ עָלַי מְאֹד לְטוֹבָה.
197. הִשְׁקַעְתִּי הַרְבֵּה מַאֲמָץ בַּלִּמּוּדִים הָאֵלֶּה.
198. אֲנִי מַצִּיעָה פִּתָּרוֹן חָכָם לַבְּעָיָה.
199. הַמְּנַהֵל הִצִּיעַ לִי מִשְׂרָה מְעוּלָּה.
200. הַזַּמָּר מוֹפִיעַ בַּתֵּאַטְרוֹן הַיּוֹם בָּעֶרֶב.
201. הִיא הוֹפִיעָה בַּטֶּלֶוִיזְיָה בַּחֲדָשׁוֹת אֶתְמוֹל.
202. אֲנִי מַבְטִיחָה לַעֲזוֹר לְךָ תָּמִיד.
203. הִבְטַחְתִּי לָבוֹא בַּזְּמַן לַפְּגִישָׁה.
204. עוֹרֵךְ-הַדִּין מוֹכִיחַ אֶת הָאֱמֶת בַּמִּשְׁפָּט.
205. הוֹכַחְתִּי אֶת הַטְּעָנָה שֶׁלִּי בְּבֵרוּר.
206. אֲנַחְנוּ מִתְמוֹדְדִים עִם הַקְּשָׁיִים בְּיַחַד.
207. הִתְמוֹדַדְתִּי הֵיטֵב עִם כָּל הַלַּחַץ.
208. הִתְנַדַּבְתִּי בַּקְּהִלָּה שָׁנָה שְׁלֵמָה בְּאַהֲבָה.
209. הִתְנַהַגְתָּ כְּמוֹ בֶּן-אָדָם אֲמִתִּי בַּמַּצָּב.
210. הָעִיר מִתְפַּתַּחַת בִּמְהִירוּת גְּדוֹלָה מְאֹד.
211. הַחֶבְרָה הִתְפַּתְּחָה מְאֹד בַּשָּׁנִים הָאַחֲרוֹנוֹת.
212. אֲנָשִׁים מִתְלוֹנְנִים עַל הַמֶּזֶג-הָאֲוִיר בַּקַּיִץ.
213. הַלָּקוֹחַ הִתְלוֹנֵן עַל הַשֵּׁרוּת הָרַע.
214. טִפַּלְתִּי בַּחוֹלֶה כָּל הַיּוֹם.
215. הִיא רוֹחֶצֶת אֶת הַיָּדַיִם בְּמַיִם.
216. רָחַצְתִּי אֶת הַפָּנִים בַּבֹּקֶר.
217. שָׁטַפְתִּי כֵּלִים אַחֲרֵי הָאֲרוּחָה.
218. הִלְבַּשְׁתִּי אֶת הַסָּבָא בְּחוּלְצָה.
219. הִיא מַפְשִׁיטָה אֶת הַמְּעִיל הַכָּבֵד.
220. הִפְשַׁטְתִּי אֶת הַסְּוֶודֶר הַחַם.
221. הֶאֱכַלְתִּי אֶת הַחוֹלֶה בַּצָּהֳרַיִם.
222. הִשְׁקֵיתִי אֶת הַסָּבָא בְּתֵה.
223. הִשְׁכַּבְתִּי אוֹתוֹ לִישׁוֹן בָּעֶרֶב.
224. הֶעֱמַדְתִּי אֶת הַכִּסֵּא לְיַד הַמִּטָּה.
225. הַמְּטַפֵּל מְסַיֵּעַ לַקָּשִׁישׁ בַּהֲלִיכָה.
226. סִיַּעְתִּי לַמִּשְׁפָּחָה בְּכָל הַסִּדּוּרִים.
227. תָּמַכְתִּי בַּסָּבָא עַד הַחֶדֶר.
228. חִמַּמְתִּי מַיִם חַמִּים לַתֵּה.
229. מָדַדְתִּי חֹם בַּבֹּקֶר בַּמִּרְפָּאָה.
230. הַכַּדּוּר הַזֶּה מְשַׁכֵּךְ כְּאֵבִים מַהֵר.
231. שִׁכַּכְתִּי אֶת הַכְּאֵב בְּעֶזְרַת תְּרוּפָה.
232. הִרְגַּעְתִּי אוֹתוֹ בִּדְבָרִים טוֹבִים.
233. הֶחֱזַרְתִּי אֶת הַסֵּפֶר לָאֲרוֹן.
234. דִּוַּחְתִּי לָרוֹפֵא עַל שִׁנּוּי.
235. הִיא מַחֲלִיפָה אֶת הַמַּצָּעִים הַיּוֹם.
236. הֶחֱלַפְתִּי אֶת הַתַּחְבּוֹשֶׁת בַּבֹּקֶר.
237. עוֹדַדְתִּי אוֹתוֹ בְּחִיּוּךְ רָחָב.
238. לִוִּיתִי אוֹתוֹ לְקוּפַּת חוֹלִים.
239. הַגַּנֶּנֶת מַשְׁגִּיחָה עַל הַיְּלָדִים בֶּחָצֵר.
240. הִשְׁגַּחְתִּי עֲלֵיהֶם כָּל הַיּוֹם.
241. אִמָּא מְחַבֶּקֶת אֶת הַיֶּלֶד בְּחֹם.
242. חִבַּקְתִּי אוֹתוֹ כְּשֶׁהוּא בָּכָה.
243. נִחַמְתִּי אֶת הַיַּלְדָּה בַּגַּן.
244. דָּנִי מְשַׁתֵּף אֶת כּוּלָּם בַּמִּשְׂחָק.
245. שִׁתַּפְתִּי אֶת הַחֲבֵרִים בַּצַּעֲצוּעִים.
246. גָּזַרְתִּי צוּרוֹת יָפוֹת מִבְּרִיסְטוֹל.
247. הִרְכַּבְתִּי מִגְדָּל גָּבוֹהַּ מִלֶּגוֹ.
248. הַגַּנֶּנֶת מַקְרִיאָה סִפּוּר לַיְּלָדִים.
249. הִקְרֵאתִי שִׁיר יָפֶה לָהֶם.
250. שָׁכַבְתִּי לָנוּחַ בַּצָּהֳרַיִם הַיּוֹם.
251. נָעַלְתִּי סַנְדָּלִים נוֹחִים לַגַּן.
252. הַיְּלָדִים רוֹחֲצִים יָדַיִם בְּסַבּוֹן.
253. רָחַצְנוּ יָדַיִם לִפְנֵי הָאוֹכֶל.
254. דָּנִי מְוַתֵּר עַל הַתּוֹר בִּנְדִיבוּת.
255. וִתַּרְתִּי לַחֲבֵרָה שֶׁלִּי בַּמִּשְׂחָק.
256. הִיא מְחַלֶּקֶת צְבָעִים לַיְּלָדִים.
257. חִלַּקְתִּי אֶת הָעוּגָה שָׁוֶה.
258. כָּל הַיְּלָדִים שָׁרִים בַּמַּעְגָּל.
259. שַׁרְנוּ שִׁיר שָׂמֵחַ בִּמְסִבָּה.
260. הַיְּלָדִים רוֹקְדִים בְּשִׂמְחָה עַכְשָׁיו.
261. רָקַדְתִּי עִם הַחֲבֵרִים בַּגַּן.
262. הִיא גּוֹמֶלֶת אוֹתוֹ מֵחִתּוּלִים עַכְשָׁיו.
263. גָּמַלְנוּ אֶת הַיֶּלֶד מִמּוֹצֵץ.
264. הָרוֹפֵא מְאַבְחֵן אֶת הַמַּחֲלָה.
265. אִבְחַנְתִּי דַּלֶּקֶת רֵאוֹת אֶתְמוֹל.
266. הָרוֹפֵא רוֹשֵׁם מִרְשָׁם לַחוֹלֶה.
267. רָשַׁמְתִּי תְּרוּפָה חֲדָשָׁה לַמְּטוּפָּל.
268. הִפְנֵיתִי אֶת הַחוֹלֶה לְמוּמְחֶה.

584. אֲנִי מַזְמִינָה מָרָק עוֹף.
585. יֵשׁ פֹּה דָּג טָרִי?

587. אֲנִי מְכִינָה סָלָט יִשְׂרְאֵלִי.
588. יֵשׁ פֹּה שֶׁמֶן זַיִת?
589. אֶפְשָׁר קְצָת מֶלַח, בְּבַקָּשָׁה?

592. אֲנִי רוֹצָה קַבָּלָה, בְּבַקָּשָׁה.
593. סְלִיחָה, כַּמָּה זֶה עוֹלֶה?
594. אַתְּ רוֹצָה עוֹד מַשֶּׁהוּ?
595. זֹאת מִסְעָדָה טוֹבָה מְאוֹד.
596. הָאוֹכֶל כָּאן טָעִים מְאוֹד.
597. הַחוּמוּס הַזֶּה טָעִים מְאוֹד.
598. הַפָלָאפֶל כָּאן טָעִים מְאוֹד.
599. אֶפְשָׁר תַּפְרִיט, בְּבַקָּשָׁה?
600. הַיּוֹם מֶזֶג אֲוִויר נָעִים.
601. הַיּוֹם יוֹרֵד גֶּשֶׁם חָזָק.
602. הַשֶּׁמֶשׁ חַמָּה מְאוֹד הַיּוֹם.
603. הַיּוֹם יֵשׁ רוּחַ חֲזָקָה.
604. הַקַּיִץ בְּיִשְׂרָאֵל חַם מְאוֹד.
605. הַחוֹרֶף פֹּה קַר מְאוֹד.
606. הָאָבִיב בְּיִשְׂרָאֵל יָפֶה מְאוֹד.
607. הַסְּתָיו בְּיִשְׂרָאֵל נָעִים מְאוֹד.
608. זֶה בֶּגֶד יָפֶה מְאֹד.
609. הַחוּלְצָה הַזֹּאת יָפָה מְאֹד.
610. אֲנִי לוֹבֶשֶׁת מִכְנָסַיִּים שְׁחוֹרִים.
611. יֵשׁ לָהּ שִׂמְלָה אֲדוּמָּה.
612. הַנַּעֲלַיִּים הָאֵלֶּה נוֹחוֹת מְאֹד.
613. אֵיזֶה צֶבַע אַתְּ אוֹהֶבֶת?
614. אֵיזוֹ מִידָּה אַתְּ צְרִיכָה?
615. אֵיפֹה תָּא הַמְּדִידָה כָּאן?
616. אֲנִי הוֹלֶכֶת לָרוֹפֵא הַיּוֹם.
617. הַמִּרְפָּאָה פְּתוּחָה עַכְשָׁיו.
618. אֵיפֹה קוּפַּת חוֹלִים שֶׁלְּךָ?
619. כּוֹאֵב לִי הָרֹאשׁ מְאֹד.
620. כּוֹאֶבֶת לִי הַבֶּטֶן הַיּוֹם.
621. כּוֹאֵב לִי הַגָּרוֹן מְאֹד.
622. יֵשׁ לִי חוֹם גָּבוֹהַּ.
623. אֲנִי לוֹקַחַת תְּרוּפָה עַכְשָׁיו.
624. אֵיפֹה יֵשׁ בֵּית מִרְקַחַת?
625. זֶה הַיּוֹם-יוֹם שֶׁלָּנוּ פֹּה.
626. זֶה הֶרְגֵּל טוֹב מְאֹד.
627. זֶה סֵדֶר יוֹם טוֹב.
628. אֵין לִי זְמַן הַיּוֹם.
629. יֵשׁ לִי הַרְבֵּה סַבְלָנוּת.
630. כַּמָּה כֶּסֶף יֵשׁ לְךָ?
631. יֵשׁ לָנוּ רֶכֶב חָדָשׁ.
632. יֵשׁ לְךָ דַּרְכּוֹן יִשְׂרְאֵלִי?

634. יֵשׁ לִי שְׁאֵלָה טוֹבָה.
635. זֹאת תְּשׁוּבָה נְכוֹנָה מְאֹד.
636. אֵין שׁוּם בְּעָיָה כָּאן.
637. זֶה מִקְצוֹעַ מְעַנְיֵן מְאֹד.
638. הַמְּנַהֵל יוֹשֵׁב בַּמִּשְׂרָד עַכְשָׁיו.
639. זֹאת חֶבְרָה גְּדוֹלָה מְאֹד.

643. יֵשׁ כָּאן מַשְׂכּוֹרֶת טוֹבָה.
644. יֵשׁ לִי הַרְבֵּה נִיסָּיוֹן.

646. יֵשׁ לִי הוֹבִּי חָדָשׁ.
647. אֵין לִי שְׁעוֹת פְּנַאי.
648. אֲנִי אוֹהֶבֶת סְפּוֹרְט מְאוֹד.
649. זֶה סֶרֶט יָפֶה מְאוֹד.
650. אֲנַחְנוּ הוֹלְכִים לַקּוֹלְנוֹעַ עַכְשָׁיו.
651. יֵשׁ לָנוּ טִיּוּל מָחָר.
652. אֲנִי אוֹהֶבֶת שְׂחִיָּה בַּיָּם.
653. יֵשׁ בְּרֵיכָה גְּדוֹלָה בַּמָּלוֹן.
654. אֲנַחְנוּ הוֹלְכִים לַיָּם הַיּוֹם.
655. הַבַּנְק פָּתוּחַ עַד אַרְבַּע.
656. אֲנִי הוֹלֶכֶת לַדּוֹאַר הַיּוֹם.
657. אֲנִי פּוֹתַחַת חֶשְׁבּוֹן בַּנְק.
658. יֵשׁ כַּסְפּוֹמָט לְיַד הַבַּנְק.
659. אֲנִי רוֹצָה לְשַׁלֵּם בִּמְזוּמָּן.
660. יֵשׁ לְךָ חֲבִילָה בַּדּוֹאַר.
661. יֵשׁ תּוֹר אָרוֹךְ בַּדּוֹאַר.
662. יֵשׁ לִי הַזְמָנָה לַמְּסִיבָּה.
663. כָּל הַתְחָלָה הִיא קָשָׁה.
664. יֵשׁ לִי הַרְגָּשָׁה טוֹבָה.
665. יֵשׁ לָנוּ חֲתוּנָּה הַיּוֹם.
666. זֹאת מַתָּנָה יָפָה מְאוֹד.
667. יוֹם הוּלֶּדֶת שָׂמֵחַ לְךָ!
668. יֵשׁ מְסִיבָּה גְּדוֹלָה הָעֶרֶב.
669. אֲנִי כּוֹתֶבֶת בְּרָכָה לַחֲתוּנָּה.
670. קָנִיתִי פְּרָחִים יָפִים לְשַׁבָּת.
671. יֵשׁ לָנוּ אוֹרֵחַ חָשׁוּב.
672. אֲנַחְנוּ שׁוֹתִים יַיִן אָדוֹם.
673. שַׁבָּת שָׁלוֹם לְכָל הַמִּשְׁפָּחָה!
674. חַג שָׂמֵחַ לְכָל הַחֲבֵרִים!
675. אִמָּא אוֹפָה חַלָּה לְשַׁבָּת.
676. הִיא מַדְלִיקָה נֵרוֹת שַׁבָּת.
677. אַבָּא עוֹשֶׂה קִידּוּשׁ בָּעֶרֶב.
678. אֲנַחְנוּ אוֹכְלִים מַצָּה בְּפֶסַח.
679. חַג רֹאשׁ הַשָּׁנָה שָׂמֵחַ!
680. חַג חֲנוּכָּה שָׂמֵחַ לְכוּלָּם!
681. בְּתֵיאָבוֹן לְכָל הַמִּשְׁפָּחָה!
682. הַיּוֹם קַר מְאֹד בַּחוּץ.
683. בְּדֶרֶךְ כְּלָל אֲנַחְנוּ לוֹמְדִים.
684. לִפְעָמִים אֲנַחְנוּ הוֹלְכִים לַיָּם.
685. אַף פַּעַם לֹא מְאוּחָר!
686. אֲנִי קָמָה מוּקְדָם בַּבֹּקֶר.
687. אֲנִי לוֹמֶדֶת כָּל יוֹם.
688. יֵשׁ לִי שְׁאֵלָה אַחַת.
689. אֵין לִי זְמַן הַיּוֹם.
690. אֵין בְּעָיָה, נְדַבֵּר מָחָר.
691. אֲנִי מְחַפֶּשֶׂת עֲבוֹדָה טוֹבָה.
692. בְּהַצְלָחָה רַבָּה בַּעֲבוֹדָה הַחֲדָשָׁה!
693. הָרַכֶּבֶת נוֹסַעַת מַהֵר מְאֹד.
694. אֶפְשָׁר בְּבַקָּשָׁה לְדַבֵּר לְאַט?
695. אֲנִי מַרְגִּישָׁה מְצוּיָן הַיּוֹם.
696. שָׁלוֹם, מְאֹד נָעִים לְהַכִּיר!
697. סְלִיחָה, אֲנִי מִצְטַעֶרֶת מְאֹד.
698. אֲנִי רוֹצָה לְדַבֵּר אִיתְּךָ.
699. דָּנָה לוֹמֶדֶת עִבְרִית אִיתּוֹ.
700. אַתְּ רוֹצָה לָבוֹא אִיתָּנוּ?
701. אֲנִי שׁוֹמַעַת אוֹתְךָ מְצוּיָן.
702. אֲנִי מַכִּירָה אוֹתוֹ טוֹב.
703. הַמּוֹרֶה רוֹאֶה אוֹתָנוּ כָּאן.
704. אֲנַחְנוּ לוֹמְדִים יַחַד בָּאוּלְפָּן.
705. בֶּטַח, אֲנִי בָּאָה מָחָר!
706. מַזָּל טוֹב לְיוֹם הַהוּלֶּדֶת!
707. לְחַיִּים, חֲבֵרִים, כּוּלָּם יַחַד!
708. עַד מֵאָה וְעֶשְׂרִים שָׁנָה!
709. שַׁבָּת שָׁלוֹם לְכָל הַמִּשְׁפָּחָה!
710. חַג שָׂמֵחַ לְכוּלָּם הַיּוֹם!
711. שָׁבוּעַ טוֹב לְכָל הַחֲבֵרִים!
712. אֲנַחְנוּ גָּרִים בְּבֵית מָלוֹן.
713. אֲנִי שׁוֹתָה מִיץ תַּפּוּזִים.
714. יֵשׁ לִי שׁוֹקוֹלָד טוֹב.
715. בַּעַל הַדִּירָה טוֹב מְאוֹד.
716. הַשּׁוּתָף שֶׁלִּי נֶחְמָד.
717. שִׁלַּמְתִּי בְּהַמְחָאָה.
718. הַמַּזְגָּן לֹא עוֹבֵד.
719. הַשֵּׂיעָר שֶׁלָּהּ יָפֶה.
720. הָעֵינַיִּים שֶׁלּוֹ כְּחוּלוֹת.
721. יֵשׁ לָהּ אוֹפִי חָזָק.
722. אֵיפֹה הַמַּפְתֵּחַ שֶׁלִּי?
723. הַנּוֹף פֹּה יָפֶה מְאוֹד.
724. הַמִּזְוָודָה שֶׁלִּי כְּבֵדָה.
725. אֵין לִי מַגֶּבֶת בַּחֶדֶר.
726. הַמַּעֲלִית לֹא עוֹבֶדֶת.
727. אֲנִי גָּרָה בַּקּוֹמָה הַשְּׁלִישִׁית.
728. קָרָאתִי עִיתּוֹן הַיּוֹם.
729. קִיבַּלְתִּי הוֹדָעָה חֲדָשָׁה.
730. סִגְרִי אֶת הַדֶּלֶת בְּבַקָּשָׁה.
731. הָרוֹקֵחַ נָתַן לִי תְּרוּפָה.
732. יֵשׁ לִי מִרְשָׁם מֵהָרוֹפֵא.
733. קָחִי שְׁנֵי כַּדּוּרִים בַּיּוֹם.
734. הַטִּיפּוֹת הֵן לָעֵינַיִּים.
735. שִׂימִי מִשְׁחָה עַל הַיָּד.
736. צָרִיךְ פְּלַסְטֶר לָאֶצְבַּע.
737. אֲנִי צָרִיךְ מְשַׁכֶּכֶת כְּאֵבִים.
738. הָלַכְנוּ לַקַּנְיוֹן הַיּוֹם.
739. הַמּוּצָר הַזֶּה זוֹל.
740. עָמַדְתִּי בְּתוֹר לַקּוּפָּה.
741. הַקּוֹנֶה שָׁאַל שְׁאֵלָה.
742. קִיבַּלְתִּי זִיכּוּי מֵהַחֲנוּת.
743. יֵשׁ לִי רַב קָו.
744. הָרַכֶּבֶת בָּאָה לָרָצִיף.
745. רָאִיתִי אֶת לוּחַ הַזְּמַנִּים.
746. הַנְּסִיעָה הָיְתָה נְעִימָה.
747. הַנּוֹסֵעַ יָשַׁב בָּאוֹטוֹבּוּס.
748. הַנְּהָג נָסַע מַהֵר.
749. הַחֲנוּת בַּפִּנָּה.
750. עֲצוֹר בַּצּוֹמֶת.
751. הָרַמְזוֹר אָדֹם.
752. הַגֶּשֶׁר גָּדוֹל וְיָפֶה.
753. הָלַכְנוּ לַמֶּרְכָּז קְנִיּוֹת.
754. עָלִיתִי בַּמַּדְרֵגוֹת.
755. הַכְּנִיסָה פְּתוּחָה.
756. אֵיפֹה הַיְּצִיאָה?
757. הָלַכְנוּ לַתֵּאַטְרוֹן אֶמֶשׁ.
758. הָאָמָּן צָיַר תְּמוּנָה יָפָה.
759. קָנִינוּ כַּרְטִיסֵי כְּנִיסָה לַמּוּזֵיאוֹן.
760. הַמַּדְרִיךְ הִסְבִּיר הַרְבֵּה.
761. יֵשׁ לִי מַתְכּוֹן טוֹב.
762. הוֹסַפְתִּי פִּלְפֵּל לַמָּרָק.
763. הַמַּחֲבַת חַמָּה.
