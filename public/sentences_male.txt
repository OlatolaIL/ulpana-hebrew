1. אֲנִי רוֹצֶה מַיִם קָרִים.
2. הוּא שׁוֹתֶה תֵּה בַּבֹּקֶר.
3. אֲנִי אוֹכֵל לֶחֶם וּגְבִינָה.
4. הוּא אָכַל בַּבַּיִת.
5. דָּוִד הוֹלֵךְ לָאוּלְפָּן.
6. אֲנִי קוֹרֵא סֵפֶר בָּעֶרֶב.
7. הוּא קָרָא אֶת הַחֲדָשׁוֹת.
8. הוּא עוֹבֵד בְּמִשְׂרָד.
9. אַתָּה יוֹדֵעַ אֶת הַתְּשׁוּבָה?
10. אֲנִי רוֹאֶה אֶת הַיָּם.
11. דָּוִד בָּא אֶתְמוֹל לָאוּלְפָּן.
12. מָה אַתָּה עוֹשֶׂה הַיּוֹם?
13. אֲנִי שׁוֹמֵעַ מוּזִיקָה.
14. הוּא פּוֹגֵשׁ אֶת הַחֲבֵר.
15. הוּא פּוֹתֵחַ אֶת הַדֶּלֶת.
16. הוּא סוֹגֵר אֶת הַחַלּוֹן.
17. הוּא עוֹמֵד לְיַד הַתַּחֲנָה.
18. אֲנִי לוֹקֵחַ תַּפּוּחַ וּמַיִם.
19. הוּא אוֹמֵר שָׁלוֹם וְתוֹדָה.
20. דָּוִד נוֹסֵעַ בָּאוֹטוֹבּוּס לָעֲבוֹדָה.
21. אֲנִי חוֹזֵר הַבַּיְתָה בָּעֶרֶב.
22. הוּא חָזַר מֵהָעֲבוֹדָה עָיֵף.
23. אֲנִי עוֹזֵר לַחֲבֵר בָּאוּלְפָּן.
24. אֲנִי זוֹכֵר אֶת כָּל הַמִּלִּים.
25. הוּא שׁוֹכֵחַ אֶת הַמַּפְתֵּחוֹת שׁוּב.
26. אֲנִי חוֹשֵׁב עַל הָעֲבוֹדָה.
27. דָּוִד מוֹצֵא אֶת הָרְחוֹב בְּקַלּוּת.
28. אֲנִי אוֹהֵב קָפֶה חָזָק בַּבֹּקֶר.
29. הוּא שָׂם סוּכָּר בַּתֵּה שֶׁלּוֹ.
30. אֲנִי קָם מוּקְדָּם בַּבֹּקֶר תָּמִיד.
31. הַיֶּלֶד יָשֵׁן עַכְשָׁיו בַּחֶדֶר.
32. דָּוִד מָכַר אֶת הַמְּכוֹנִית שֶׁלּוֹ.
33. אֲנִי בּוֹחֵר יְרָקוֹת טְרִיִּים בַּחֲנוּת.
34. הוּא שׁוֹמֵר עַל הַכֶּסֶף בַּבַּנְק.
35. דָּוִד עוֹבֵר לְדִירָה חֲדָשָׁה הַחֹדֶשׁ.
36. אֲנִי מְדַבֵּר עִם הַמּוֹרֶה עִבְרִית.
37. הוּא מְבַקֵּשׁ עֶזְרָה מֵהַחֲבֵר.
38. אֲנִי מְשַׁלֵּם בְּכַרְטִיס אַשְׁרַאי עַכְשָׁיו.
39. אֲנִי מְחַפֵּשׂ דִּירָה בְּתֵל-אָבִיב.
40. אֲנִי מְקַבֵּל חֲבִילָה מֵהַדֹּאַר הַיּוֹם.
41. אֲנִי מְבַשֵּׁל מָרָק טָעִים בַּמִּטְבָּח.
42. דָּוִד מְנַקֶּה אֶת הַשּׁוּלְחָן בַּסָּלוֹן.
43. אֲנִי מְצַלֵּם אֶת הַנּוֹף הַיָּפֶה.
44. בֶּעָבָר הוּא עִשֵּׁן סִיגַרְיוֹת הַרְבֵּה.
45. אֲנִי מְחַכֶּה לָאוֹטוֹבּוּס בַּתַּחֲנָה הַמֶּרְכָּזִית.
46. אֲנִי מְקַוֶּוה לְטוֹב תָּמִיד בַּחַיִּים.
47. הוּא מְשַׁנֶּה אֶת הַתּוֹכְנִית שֶׁלּוֹ.
48. אֲנִי מְנַסֶּה לְדַבֵּר עִבְרִית.
49. אֲנִי מְאַחֵל לְךָ הַצְלָחָה רַבָּה.
50. דָּוִד מְאַרְגֵּן מְסִבָּה יָפָה בָּעֶרֶב.
51. אֲנִי מַרְגִּישׁ מְצוּיָּן הַיּוֹם בַּבֹּקֶר.
52. אֲנִי מַזְמִין פִּיצָה חַמָּה לַבַּיִת.
53. אֲנִי מֵבִין אֶת הַשְּׁאֵלָה הֵיטֵב.
54. אֲנִי מַכִּיר אֶת הָאִישׁ הַזֶּה.
55. דָּוִד מֵבִיא עוּגָה טְעִימָה לַמְּסִבָּה.
56. אֲנִי מַדְלִיק אֶת הָאוֹר בַּחֶדֶר.
57. אֲנִי מַעֲדִיף תֵּה עַל קָפֶה.
58. אֲנִי מַסְכִּים עִם כָּל מִלָּה.
59. דָּוִד מִתְרַחֵץ בַּמַּיִם חַמִּים בַּבֹּקֶר.
60. הַיֶּלֶד מִתְלַבֵּשׁ לְבַד יָפֶה מְאֹד.
61. אֲנִי מִתְרַגֵּשׁ מֵהַפְּגִישָׁה הַחֲדָשָׁה הַיּוֹם.
62. אֲנִי מִשְׁתַּמֵּשׁ בַּמַּחְשֵׁב בָּעֲבוֹדָה תָּמִיד.
63. דָּוִד הִתְחַתֵּן עִם שָׂרָה.
64. אֲנִי מִצְטַעֵר עַל הַטָּעוּת הַזֹּאת.
65. אֲנִי מִתְקַשֵּׁר לְאִמָּא בָּעֶרֶב.
66. אֲנִי מִתְרַגֵּל לַמֶּזֶג הָאֲוִיר בָּאָרֶץ.
67. אֲנִי נִשְׁאָר בַּבַּיִת הַיּוֹם בָּעֶרֶב.
68. הוּא נִזְהָר מְאֹד בַּכְּבִישׁ הַסּוֹאֵן.
69. אֲנִי נִזְכָּר בַּסִּפּוּר הַיָּפֶה עַכְשָׁיו.
70. אֲנִי מַסְפִּיק לָרַכֶּבֶת בַּבֹּקֶר תָּמִיד.
71. הוּא מַשְׁקִיעַ בַּעֲסָקִים חֲדָשִׁים בָּאָרֶץ.
72. אֲנִי מַצִּיעַ פִּתָּרוֹן חָכָם לַבְּעָיָה.
73. אֲנִי מַבְטִיחַ לַעֲזוֹר לְךָ תָּמִיד.
74. הוּא מִתְנַדֵּב בְּמָדָ״א כָּל שָׁבוּעַ.
75. הַיֶּלֶד מִתְנַהֵג יָפֶה מְאֹד בַּכִּתָּה.
76. הוּא מְטַפֵּל בַּסָּבָא הַיּוֹם.
77. הוּא שׁוֹטֵף אֶת הָרִצְפָּה כָּעֵת.
78. הוּא מַלְבִּישׁ אֶת הַמְּטוּפָּל בִּזְהִירוּת.
79. הוּא מַאֲכִיל אֶת הַסָּבָא בְּמָרָק.
80. הוּא מַשְׁקֶה אוֹתוֹ בְּמַיִם קָרִים.
81. הוּא מַשְׁכִּיב אֶת הַחוֹלֶה בַּמִּטָּה.
82. הוּא מַעֲמִיד אוֹתוֹ עַל הָרַגְלַיִם.
83. הוּא תּוֹמֵךְ בּוֹ בִּזְמַן הַהֲלִיכָה.
84. הוּא מְחַמֵּם אֶת הָאוֹכֶל בַּמִּיקְרוֹגַל.
85. הוּא מוֹדֵד לָהּ לַחַץ דָּם.
86. הוּא מַרְגִּיעַ אֶת הַחוֹלֶה הַמְּבֹהָל.
87. הוּא מַחֲזִיר הַכֹּל לַמָּקוֹם.
88. הוּא מְדַוֵּחַ עַל הַמַּצָּב לָאָחוֹת.
89. הוּא מְעוֹדֵד אֶת הַסָּבָא תָּמִיד.
90. הוּא מְלַוֶּה אֶת הַקָּשִׁישׁ לַבַּנְק.
91. הוּא מְנַחֵם אֶת הַחָבֵר הֶעָצוּב.
92. הַיֶּלֶד גּוֹזֵר נְיָר בְּמִסְפָּרַיִם.
93. יוֹסִי מַרְכִּיב פָּאזֶל גָּדוֹל הַיּוֹם.
94. הַיֶּלֶד שׁוֹכֵב עַל הַמִּזְרָן.
95. הוּא נוֹעֵל נַעֲלַיִם לְבַד הַיּוֹם.
96. הוּא מַפְנֶה אוֹתִי לִבְדִיקוֹת.
97. הוּא חוֹבֵשׁ אֶת הַיָּד הַפְּצוּעָה.
98. הוּא מַרְדִּים לִפְנֵי הַנִּתּוּחַ.
99. הוּא מַחְלִים מֵהַמַּחֲלָה בְּקָרוֹב.
100. הוּא מִשְׁתַּעֵל בְּלִי הַפְסָקָה הַיּוֹם.
101. הַיֶּלֶד מֵקִיא מֵהַרְעָלַת קֵבָה.
102. הוּא נוֹשֵׁם עָמֹק בַּבְּדִיקָה.
103. הוּא בּוֹלֵעַ כַּדּוּר עִם מַיִם.
104. הוּא מְמַשֵּׁשׁ אֶת הַבֶּטֶן.
105. הוּא סוֹבֵל מִכְּאֵב רֹאשׁ.
106. הוּא מֵפִיק חֶשְׁבּוֹנִית מַס מְהִירָה.
107. הוּא מְקַזֵּז אֶת הַמַּס הַחֹדֶשׁ.
108. הוּא מַפְקִיד כֶּסֶף בַּבַּנְק כָּעֵת.
109. הוּא מוֹשֵׁךְ כֶּסֶף מֵהַכַּסְפּוֹמָט.
110. הוּא מַעֲבִיר כֶּסֶף לַסַּפָּק.
111. הוּא מְנַהֵל סִפְרֵי חֶשְׁבּוֹנוֹת.
112. הוּא עוֹקֵב אַחַר כָּל הוֹצָאָה.
113. הוּא פּוֹרֵס לְתַשְׁלוּמִים שְׁוִוים.
114. הוּא מְבַטֵּל חֶשְׁבּוֹנִית שְׁגוּיָה כָּעֵת.
115. הוּא מַפְחִית אֶת עֵרֶךְ הַצִיּוּד.
116. הוּא מַגִּישׁ דּוּ״חַ לְמַס הַכְנָסָה.
117. הוּא מַסְדִּיר אֶת הַתַּשְׁלוּם הַיּוֹם.
118. הוּא מֵנִיעַ אֶת הָרֶכֶב בַּבֹּקֶר.
119. הוּא מְשַׁמֵּן אֶת הַחֲלָקִים.
120. הוּא מְכַוֵּן פָּרָמֶטְרִים בַּמַּחְשֵׁב.
121. הוּא מְנַקֵּז אֶת שֶׁמֶן הַבְּלָמִים.
122. הוּא מְמַלֵּא דֶּלֶק בַּתַּחֲנָה.
123. הוּא מַטְעִין אֶת הַמַּצְבֵּר.
124. הוּא מְחַבֵּר אֶת הַכְּבָלִים.
125. הוּא מְנַתֵּק אֶת הַזֶּרֶם.
126. הוּא מַהְדֵּק אֶת הַבְּרָגִים.
127. הוּא פּוֹנֶה יָמִינָה בָּרַמְזוֹר.
128. הוּא אוֹטֵם אֶת הַחִבּוּר בְּסִילִיקוֹן.
129. הוּא מְפַתֵּחַ מַפְתֵּחַ עִנְיָנִים.
130. הוּא מַאֲרִיךְ אֶת הַהַשְׁאָלָה.
131. הוּא מְאַתֵּר אֶת הַסֵּפֶר בַּמַּדָּף.
132. הוּא מְשַׁקֵּם דַּף קָרוּעַ בִּזְהִירוּת.
133. הוּא סוֹרֵק תְּמוּנוֹת הִיסְטוֹרִיּוֹת.
134. הוּא גּוֹנֵז מִסְמָכִים בְּאַכְסַנְיָה.
135. הוּא כּוֹרֵךְ סֵפֶר בִּכְרִיכָה קָשָׁה.
136. הוּא מְתַעֵד כָּל תְּנוּעָה בָּאוֹסֶף.
137. הוּא שׁוֹלֵף תִּיק מֵהַמַּחְשֵׁב.
138. הוּא מְסַוֵּג אֶת הַחוֹמֶר.
139. הוּא מַנְפִּיק כַּרְטִיס קוֹרֵא לַתַּלְמִיד.
140. הוּא מַדְפִּיס דַּף בַּמַּדְפֶּסֶת.
141. הוּא מַקְצִיף אֶת כָּל הָרֶכֶב.
142. הוּא מְנַגֵּב חַלּוֹנוֹת בְּמַטְלִית.
143. הוּא שׁוֹאֵב אֶת הַשָּׁטִיחַ בָּרֶכֶב.
144. הוּא מַבְרִיק אֶת הָרֶכֶב.
145. הוּא מוֹרֵחַ וַקְס עַל הַמִּכְסֶה.
146. הוּא מְיַבֵּשׁ אֶת הָרֶכֶב בַּאֲוִויר.
147. הוּא מְרַסֵּס חוֹמֶר עַל הַגַּלְגַּלִּים.
148. הוּא מְקַרְצֵף אֶת הַלִּכְלוּךְ בְּמִבְרֶשֶׁת.
149. הוּא מַתִּיז מַיִם בְּלַחַץ גָּבוֹהַּ.
150. הוּא מֵסִיר כְּתָמִים קָשִׁים מֵהָרִפּוּד.
151. הוּא מַשְׁחִיר אֶת הַצְּמִיגִים בְּחוֹמֶר.
152. הוּא מְבַשֵּׂם אֶת פְּנִים הָרֶכֶב.
153. הוּא מְרוֹקֵן אֶת פַּח הָאַשְׁפָּה.
154. הוּא מְנַעֵר אֶת שַׁטִיחֵי הָרֶכֶב.
155. הוּא מַכְנִיס אֶת הָרֶכֶב לַמִּנְהָרָה.
