1. אֲנִי רוֹצֶה מַיִם קָרִים.
2. אֶתְמוֹל רָצִיתִי קָפֶה.
3. הוּא שׁוֹתֶה תֵּה בַּבֹּקֶר.
4. אֶתְמוֹל שָׁתִיתִי מִיץ.
5. אֲנִי אוֹכֵל לֶחֶם וּגְבִינָה.
6. הוּא אָכַל בַּבַּיִת.
7. דָּוִד הוֹלֵךְ לָאוּלְפָּן.
8. אֶתְמוֹל הָלַכְתִּי לַעֲבוֹדָה.
9. הִיא כּוֹתֶבֶת מִכְתָּב לְאִמָּא.
10. כָּתַבְתִּי הוֹדָעָה קְצָרָה.
11. אֲנִי קוֹרֵא סֵפֶר בָּעֶרֶב.
12. הוּא קָרָא אֶת הַחֲדָשׁוֹת.
13. אֲנַחְנוּ לוֹמְדִים עִבְרִית בָּאוּלְפָּן.
14. לָמַדְתִּי מִלִּים חֲדָשׁוֹת.
15. דָּנִיאֵל גָּר בִּירוּשָׁלַיִם.
16. גַּרְתִּי בְּחֵיפָה שָׁנָה.
17. הִיא קוֹנָה פֵּרוֹת בַּשּׁוּק.
18. אֶתְמוֹל קָנִיתִי חָלָב.
19. הוּא עוֹבֵד בְּמִשְׂרָד.
20. עָבַדְתִּי כָּל הַיּוֹם.
21. אַתָּה יוֹדֵעַ אֶת הַתְּשׁוּבָה?
22. לֹא יָדַעְתִּי אֶת זֶה.
23. אֲנִי רוֹאֶה אֶת הַיָּם.
24. רָאִיתִי אֶת הַחֲבֵר בָּרְחוֹב.
25. הִיא בָּאָה לַבַּיִת עַכְשָׁיו.
26. דָּוִד בָּא אֶתְמוֹל לָאוּלְפָּן.
27. מָה אַתָּה עוֹשֶׂה הַיּוֹם?
28. עָשִׂיתִי שִׁעוּרֵי בַּיִת.
29. אֲנִי שׁוֹמֵעַ מוּזִיקָה.
30. שָׁמַעְתִּי אֶת הַשִּׁיר הַזֶּה.
31. הוּא פּוֹגֵשׁ אֶת הַחֲבֵר.
32. פָּגַשְׁתִּי אוֹתוֹ בַּבֹּקֶר.
33. הוּא פּוֹתֵחַ אֶת הַדֶּלֶת.
34. פָּתַחְתִּי אֶת הַחַלּוֹן.
35. הוּא סוֹגֵר אֶת הַחַלּוֹן.
36. סָגַרְתִּי אֶת הַדֶּלֶת.
37. אֲנַחְנוּ יוֹשְׁבִים בְּבֵית קָפֶה.
38. יָשַׁבְתִּי עַל הַכִּסֵּא.
39. הוּא עוֹמֵד לְיַד הַתַּחֲנָה.
40. עָמַדְתִּי בַּתּוֹר בַּסּוּפֶּר.
41. אֲנִי לוֹקֵחַ תַּפּוּחַ וּמַיִם.
42. לָקַחְתִּי אֶת הַמַּפְתֵּחַ.
43. הִיא נוֹתֶנֶת מַתָּנָה לַחֲבֵר.
44. נָתַתִּי לוֹ אֶת הַסֵּפֶר.
45. הַתַּלְמִיד שׁוֹאֵל אֶת הַמּוֹרֶה.
46. שָׁאַלְתִּי שְׁאֵלָה פְּשׁוּטָה.
47. הַמּוֹרֶה עוֹנֶה לַתַּלְמִיד.
48. עָנִיתִי עַל הַשְּׁאֵלָה.
49. הוּא אוֹמֵר שָׁלוֹם וְתוֹדָה.
50. אָמַרְתִּי לוֹ תּוֹדָה רַבָּה.
51. דָּוִד נוֹסֵעַ בָּאוֹטוֹבּוּס לָעֲבוֹדָה.
52. אֶתְמוֹל נָסַעְתִּי בָּרַכֶּבֶת לְחֵיפָה.
53. אֲנִי חוֹזֵר הַבַּיְתָה בָּעֶרֶב.
54. הוּא חָזַר מֵהָעֲבוֹדָה עָיֵף.
55. דָּנִי רָץ בַּפַּארְק בַּבֹּקֶר.
56. אֶתְמוֹל רַצְתִּי חֲמִשָּׁה קִילוֹמֶטְרִים.
57. אֲנִי עוֹזֵר לַחֲבֵר בָּאוּלְפָּן.
58. שָׂרָה עָזְרָה לְאִמָּא בַּבַּיִת.
59. אֲנִי זוֹכֵר אֶת כָּל הַמִּלִּים.
60. זָכַרְתִּי אֶת מִסְפַּר הַטֶּלֶפוֹן.
61. הוּא שׁוֹכֵחַ אֶת הַמַּפְתֵּחוֹת שׁוּב.
62. שָׁכַחְתִּי אֶת הַתִּיק בַּבַּיִת.
63. אֲנִי חוֹשֵׁב עַל הָעֲבוֹדָה.
64. אֶתְמוֹל חָשַׁבְתִּי עַל הָרַעְיוֹן.
65. דָּוִד מוֹצֵא אֶת הָרְחוֹב בְּקַלּוּת.
66. מָצָאתִי עֲבוֹדָה טוֹבָה בָּעִיר.
67. אֲנִי אוֹהֵב קָפֶה חָזָק בַּבֹּקֶר.
68. אָהַבְתִּי אֶת הַסֶּפֶר הַזֶּה מְאֹד.
69. הוּא שָׂם סוּכָּר בַּתֵּה שֶׁלּוֹ.
70. שַׂמְתִּי אֶת הַכֶּסֶף בַּתִּיק.
71. אֲנִי קָם מוּקְדָּם בַּבֹּקֶר תָּמִיד.
72. הַיּוֹם קַמְתִּי בְּשָׁעָה שֶׁבַע.
73. הַיֶּלֶד יָשֵׁן עַכְשָׁיו בַּחֶדֶר.
74. אֶתְמוֹל יָשַׁנְתִּי שְׁמוֹנֶה שָׁעוֹת טוֹבוֹת.
75. שַׂבָּא נָח בַּבַּיִת בַּצָּהֳרַיִם.
76. בַּשַּׁבָּת נַחְנוּ כָּל הַיּוֹם.
77. אֲנַחְנוּ טָסִים מָחָר לְאֵילַת.
78. בַּשָּׁנָה שֶׁעָבְרָה טַסְתִּי לְפָרִיז.
79. הַסּוֹחֵר מוֹכֵר פֵּרוֹת בַּשּׁוּק.
80. דָּוִד מָכַר אֶת הַמְּכוֹנִית שֶׁלּוֹ.
81. אֲנִי בּוֹחֵר יְרָקוֹת טְרִיִּים בַּחֲנוּת.
82. בָּחַרְתִּי מָנָה טְעִימָה בַּמִּסְעָדָה.
83. הָרוֹפֵא בּוֹדֵק אֶת הַחוֹלֶה הַיּוֹם.
84. בָּדַקְתִּי אֶת הַחֶשְׁבּוֹן בַּקְּפֵדָה.
85. הוּא שׁוֹמֵר עַל הַכֶּסֶף בַּבַּנְק.
86. שָׁמַרְתִּי לְךָ מָקוֹם טוֹב.
87. דָּוִד עוֹבֵר לְדִירָה חֲדָשָׁה הַחֹדֶשׁ.
88. אֶתְמוֹל עָבַרְנוּ אֶת הַמִּבְחָן בְּהַצְלָחָה.
89. הָאוֹטוֹבּוּס עוֹצֵר בַּתַּחֲנָה הַקְּרוֹבָה.
90. הַנַּהָג עָצַר בָּאוֹר הָאָדֹם.
91. אֲנִי מְדַבֵּר עִם הַמּוֹרֶה עִבְרִית.
92. אֶתְמוֹל דִּבַּרְתִּי עִם דָּוִד בַּטֶּלֶפוֹן.
93. הוּא מְבַקֵּשׁ עֶזְרָה מֵהַחֲבֵר.
94. בִּקַּשְׁתִּי חֶשְׁבּוֹן בַּמִּסְעָדָה בְּנִימוּס.
95. אֲנִי מְשַׁלֵּם בְּכַרְטִיס אַשְׁרַאי עַכְשָׁיו.
96. שִׁילַּמְתִּי עַל הַקָּפֶה בַּבֹּקֶר.
97. סַבָּא מְסַפֵּר סִפּוּר יָפֶה לַיְּלָדִים.
98. סִפַּרְתִּי לַחֲבֵר עַל הַטִּיּוּל.
99. אֲנַחְנוּ מְטַיְּלִים בִּירוּשָׁלַיִם הַיּוֹם בְּיַחַד.
100. בַּשַּׁבָּת טִיַּלְנוּ בַּפַּארְק הַלְּאוּמִּי.
101. אֲנִי מְחַפֵּשׂ דִּירָה בְּתֵל-אָבִיב.
102. חִפַּשְׂתִּי אֶת הַמַּפְתֵּחוֹת שָׁעָה שְׁלֵמָה.
103. הִיא מְסַדֶּרֶת אֶת הַחֶדֶר יָפֶה.
104. אֶתְמוֹל סִדַּרְתִּי אֶת הָאֲרוֹן בַּבַּיִת.
105. אֲנִי מְקַבֵּל חֲבִילָה מֵהַדֹּאַר הַיּוֹם.
106. קִבַּלְתִּי הוֹדָעָה חֲשׁוּבָה מֵהַבַּנְק.
107. הַמּוֹרֶה מְלַמֵּד עִבְרִית בָּאוּלְפָּן.
108. שָׁנָה שְׁלֵמָה לִמַּדְתִּי בְּבֵית-סֵפֶר.
109. הַיְּלָדִים מְשַׂחֲקִים בַּכַּדּוּר בַּחֲצַר.
110. אֶתְמוֹל שִׂחַקְנוּ שַׁחְמָט בַּבַּיִת.
111. אֲנִי מְבַשֵּׁל מָרָק טָעִים בַּמִּטְבָּח.
112. אִמָּא בִּשְּׁלָה אֲרוּחַת עֶרֶב נִפְלָאָה.
113. דָּוִד מְנַקֶּה אֶת הַשּׁוּלְחָן בַּסָּלוֹן.
114. אֶתְמוֹל נִקִּיתִי אֶת כָּל הַדִּירָה.
115. הַיַּלְדָּה מְצַיֶּרֶת פְּרָחִים בְּצִבְעוֹנִיּוּת.
116. צִיַּרְתִּי תְּמוּנָה יָפָה עַל נְיָר.
117. אֲנִי מְצַלֵּם אֶת הַנּוֹף הַיָּפֶה.
118. אֶתְמוֹל צִלַּמְתִּי הַרְבֵּה תְּמוּנוֹת בַּיָּם.
119. אֲנַחְנוּ מְבַקְּרִים אֵצֶל סַבָּא בָּעֶרֶב.
120. בַּשַּׁבָּת בִּקַּרְתִּי בַּמּוּזֵאוֹן בִּירוּשָׁלַיִם.
121. אָסוּר לְעַשֵּׁן בַּמָּקוֹם הַצִּבּוּרִי הַזֶּה.
122. בֶּעָבָר הוּא עִשֵּׁן סִיגַרְיוֹת הַרְבֵּה.
123. הַטֶּכְנַאי מְתַקֵּן אֶת הַמַּחְשֵׁב הַיּוֹם.
124. תִּקַּנְתִּי אֶת הַטָּעוּת בַּמִּכְתָּב.
125. אַבָּא מְבָרֵךְ עַל הַיַּיִן בְּשַׁבָּת.
126. בֵּרַכְתִּי אֶת הֶחָבֵר לְיוֹם-הוּלֶּדֶת.
127. אֲנִי מְחַכֶּה לָאוֹטוֹבּוּס בַּתַּחֲנָה הַמֶּרְכָּזִית.
128. חִכִּיתִי לְךָ עֶשְׂרִים דַּקּוֹת שְׁלֵמוֹת.
129. אֲנִי מְקַוֶּוה לְטוֹב תָּמִיד בַּחַיִּים.
130. קִוִּיתִי לִפְגּוֹשׁ אוֹתְךָ הַיּוֹם כָּאן.
131. הוּא מְשַׁנֶּה אֶת הַתּוֹכְנִית שֶׁלּוֹ.
132. שִׁנִּיתִי אֶת הַסִּסְמָה שֶׁלִּי בַּמַּחְשֵׁב.
133. אֲנִי מְנַסֶּה לְדַבֵּר עִבְרִית.
134. אֶתְמוֹל נִסִּיתִי מָנָה חֲדָשָׁה בַּמִּסְעָדָה.
135. אֲנִי מְאַחֵל לְךָ הַצְלָחָה רַבָּה.
136. אִחַלְנוּ חַג שָׂמֵחַ לְכָל הַמִּשְׁפָּחָה.
137. הִיא מְסַיֶּמֶת אֶת הָעֲבוֹדָה בַּזְּמַן.
138. סִיַּמְתִּי אֶת הַקּוּרְס בְּהַצְלָחָה גְּדוֹלָה.
139. דָּוִד מְאַרְגֵּן מְסִבָּה יָפָה בָּעֶרֶב.
140. אִרְגַּנְתִּי אֶת הַמִּסְמָכִים שֶׁלִּי הַיּוֹם.
141. אֲנִי מַרְגִּישׁ מְצוּיָּן הַיּוֹם בַּבֹּקֶר.
142. אֶתְמוֹל הִרְגַּשְׁתִּי עָיֵף מְאֹד בָּעֶרֶב.
143. אֲנִי מַזְמִין פִּיצָה חַמָּה לַבַּיִת.
144. הִזְמַנְתִּי מָקוֹם טוֹב בַּמִּסְעָדָה מֵרֹאשׁ.
145. הַשִּׁעוּר שֶׁלָּנוּ מַתְחִיל בְּשָׁעָה תֵּשַׁע.
146. הִתְחַלְתִּי לַעֲבוֹד בַּמִּשְׂרָד הֶחָדָשׁ.
147. הָרַכֶּבֶת מַגִּיעָה בִּזְמַן לַתַּחֲנָה עַכְשָׁיו.
148. הִגַּעְתִּי לָאוּלְפָּן בַּבֹּקֶר בְּשָׁלוֹם.
149. אֲנִי מֵבִין אֶת הַשְּׁאֵלָה הֵיטֵב.
150. לֹא הֵבַנְתִּי אֶת הַמִּלָּה הַזֹּאת.
151. אֲנִי מַכִּיר אֶת הָאִישׁ הַזֶּה.
152. הִכַּרְתִּי אוֹתוֹ בָּאוּלְפָּן בַּקַּיִץ.
153. הַמּוֹרֶה מַסְבִּיר אֶת הַכְּלָל בְּבֵרוּר.
154. הִסְבַּרְתִּי לַחֲבֵר אֶת הַדֶּרֶךְ.
155. הַתַּלְמִיד מַקְשִׁיב לַמּוֹרֶה בַּכִּתָּה בְּקֶשֶׁב.
156. הִקְשַׁבְתִּי לַחֲדָשׁוֹת בָּרַדְיוֹ בַּבֹּקֶר.
157. אֲנַחְנוּ מַמְשִׁיכִים לִלְמוֹד עִבְרִית בְּשִׂמְחָה.
158. הִמְשַׁכְתִּי יָשָׁר עַד הָרַמְזוֹר.
159. הַמְּנַהֵל מַחְלִיט עַל הַתּוֹכְנִית הַיּוֹם.
160. הֶחְלַטְתִּי לִנְסוֹעַ לִירוּשָׁלַיִם מָחָר.
161. דָּוִד מֵבִיא עוּגָה טְעִימָה לַמְּסִבָּה.
162. הֵבֵאתִי מַתָּנָה יָפָה מִפָּרִיז.
163. אֲנִי מַדְלִיק אֶת הָאוֹר בַּחֶדֶר.
164. אֶתְמוֹל הִדְלַקְתִּי אֶת הַמַּזְגָן בַּבַּיִת.
165. הִיא מְכַבָּה אֶת הָאוֹר בַּלַּיְלָה.
166. כִּבִּיתִי אֶת הַטֶּלֶוִיזְיָה לִפְנֵי הַשֵּׁנָה.
167. הַגֶּשֶׁם מַפְסִיק לָרֶדֶת בַּחוּץ עַכְשָׁיו.
168. הִפְסַקְתִּי לַעֲבוֹד מְאוּחָר בַּלַּיְלָה.
169. הַמֶּלְצַר מַמְלִיץ עַל הַדָּג הַטָּרִי.
170. הִמְלַצְתִּי עַל הַמָּלוֹן הַזֶּה לַחֲבֵרִים.
171. הַתַּלְמִיד מַצְלִיחַ בַּלִּמּוּדִים בָּאוּלְפָּן יָפֶה.
172. הִצְלַחְתִּי בַּמִּבְחָן הַקָּשֶׁה בְּעִבְרִית.
173. אֲנִי מַעֲדִיף תֵּה עַל קָפֶה.
174. הֶעֱדַפְתִּי לְהִשָּׁאֵר בַּבַּיִת בַּגֶּשֶׁם.
175. אֲנִי מַסְכִּים עִם כָּל מִלָּה.
176. הִסְכַּמְנוּ עַל הַמְּחִיר מִיָּד שָׁם.
177. דָּוִד מִתְרַחֵץ בַּמַּיִם חַמִּים בַּבֹּקֶר.
178. הִתְרַחַצְתִּי בַּיָּם עִם כָּל הַחֲבֵרִים.
179. הַיֶּלֶד מִתְלַבֵּשׁ לְבַד יָפֶה מְאֹד.
180. הִתְלַבַּשְׁתִּי מַהֵר בַּבֹּקֶר לָעֲבוֹדָה.
181. אֲנִי מִתְרַגֵּשׁ מֵהַפְּגִישָׁה הַחֲדָשָׁה הַיּוֹם.
182. מְאֹד הִתְרַגַּשְׁתִּי בַּחֲתוּנָּה שֶׁל אָחִי.
183. אֲנִי מִשְׁתַּמֵּשׁ בַּמַּחְשֵׁב בָּעֲבוֹדָה תָּמִיד.
184. הִשְׁתַּמַּשְׁתִּי בַּמִּלּוֹן בַּשִּׁעוּר אֶתְמוֹל.
185. הֵם מִתְחַתְּנִים בַּקַּיִץ הַקָּרוֹב בְּיִשְׂרָאֵל.
186. דָּוִד הִתְחַתֵּן עִם שָׂרָה.
187. הָאִישׁ מִתְפַּלֵּל בְּבֵית-הַכְּנֶסֶת בַּבֹּקֶר.
188. הִתְפַּלַּלְנוּ שָׁם בְּיוֹם כִּפּוּר בְּיַחַד.
189. אֲנִי מִצְטַעֵר עַל הַטָּעוּת הַזֹּאת.
190. מְאֹד הִצְטַעַרְתִּי עַל מָה שֶׁקָּרָה.
191. אֲנִי מִתְקַשֵּׁר לְאִמָּא בָּעֶרֶב.
192. אֶתְמוֹל הִתְקַשַּׁרְתִּי לָרוֹפֵא בַּבֹּקֶר.
193. אֲנַחְנוּ מִתְרָאִים שׁוּב בָּעֶרֶב בַּקָּפֶה.
194. הִתְרָאֵינוּ בַּקָּפֶה בַּשָּׁבוּעַ שֶׁעָבַר.
195. הִיא מִסְתַּכֶּלֶת עַל הַתְּמוּנָה הַיָּפָה.
196. הִסְתַּכַּלְתִּי בַּחַלּוֹן שָׁעָה אֲרוּכָּה הַיּוֹם.
197. הַתַּלְמִיד מִתְקַדֵּם יָפֶה בָּאוּלְפָּן הַשָּׁנָה.
198. הִתְקַדַּמְתִּי הַרְבֵּה בַּחֹדֶשׁ הָאַחֲרוֹן בָּעֲבוֹדָה.
199. אֲנִי מִתְרַגֵּל לַמֶּזֶג הָאֲוִיר בָּאָרֶץ.
200. הִתְרַגַּלְתִּי מַהֵר לַחַיִּים בְּיִשְׂרָאֵל.
201. הַתַּלְמִיד נִכְנָס לַכִּתָּה בְּשֶׁקֶט.
202. אֶתְמוֹל נִכְנַסְתִּי לַמִּשְׂרָד מוּקְדָּם בַּבֹּקֶר.
203. אֲנִי נִשְׁאָר בַּבַּיִת הַיּוֹם בָּעֶרֶב.
204. אֶתְמוֹל נִשְׁאַרְתִּי בָּעֲבוֹדָה עַד מְאוּחָר.
205. אֲנַחְנוּ נִפְגָּשִׁים בַּקָּפֶה בָּעֶרֶב בְּיַחַד.
206. אֶתְמוֹל נִפְגַּשְׁתִּי עִם הַמְּנַהֵל לְשִׂיחָה.
207. הַחֲנוּת נִסְגֶּרֶת בְּשָׁעָה שְׁמוֹנֶה בָּעֶרֶב.
208. הַדֶּלֶת נִסְגְּרָה לְפֶתַע פִּתְאוֹם בָּרוּחַ.
209. הַסִּפְרִיָּה נִפְתַּחַת בַּבֹּקֶר בְּשָׁעָה שְׁמוֹנֶה.
210. הַקּוּרְס הֶחָדָשׁ נִפְתַּח אֶתְמוֹל בָּאוּלְפָּן.
211. הוּא נִזְהָר מְאֹד בַּכְּבִישׁ הַסּוֹאֵן.
212. נִזְהַרְתִּי מְאֹד בַּדֶּרֶךְ הַחֲלָקָה בַּגֶּשֶׁם.
213. תִּינוֹק חָדָשׁ נוֹלָד הַיּוֹם בַּמִּשְׁפָּחָה.
214. נוֹלַדְתִּי בְּיִשְׂרָאֵל לִפְנֵי עֶשְׂרִים שָׁנָה.
215. אֲנִי נִזְכָּר בַּסִּפּוּר הַיָּפֶה עַכְשָׁיו.
216. פִּתְאוֹם נִזְכַּרְתִּי בַּשֵּׁם שֶׁלּוֹ מִיָּד.
217. קָשֶׁה לְהִפָּרֵד מֵחֲבֵרִים טוֹבִים תָּמִיד.
218. אֶתְמוֹל נִפְרַדְנוּ בִּשְׂדֵה הַתְּעוּפָה בְּחִבּוּק.
219. הַשֶּׁמֶשׁ נֶעֱלֶמֶת מֵאֲחוֹרֵי הֶעָנָן לְאַט.
220. הַמַּפְתֵּחַ שֶׁלִּי נֶעֱלַם בַּבֹּקֶר פִּתְאוֹם.
221. הַתִּינוֹק נִרְדָּם בָּעֶרֶשׂ בְּשֶׁקֶט עַכְשָׁיו.
222. נִרְדַּמְתִּי מַהֵר מוּל הַטֶּלֶוִיזְיָה אֶתְמוֹל.
223. הַסֶּרֶט נִמְשָׁךְ שְׁעָתַיִם תְּמִימוֹת בַּקּוֹלְנוֹעַ.
224. הַיְּשִׁיבָה נִמְשְׁכָה הַרְבֵּה זְמַן אֶתְמוֹל.
225. אַף אֶחָד לֹא נִכְשָׁל כָּאן.
226. נִכְשַׁלְתִּי בַּטֶּסְט הָרִאשׁוֹן שֶׁל נְהִיגָה.
227. אֲנִי מַסְפִּיק לָרַכֶּבֶת בַּבֹּקֶר תָּמִיד.
228. הִסְפַּקְתִּי לְסַיֵּם אֶת הַדּוּ״חַ הַיּוֹם.
229. הַמַּצָּב הַכַּלְכָּלִי מַשְׁפִּיעַ עַל הַכֹּל.
230. הַמּוֹרֶה הִשְׁפִּיעַ עָלַי מְאֹד לְטוֹבָה.
231. הוּא מַשְׁקִיעַ בַּעֲסָקִים חֲדָשִׁים בָּאָרֶץ.
232. הִשְׁקַעְתִּי הַרְבֵּה מַאֲמָץ בַּלִּמּוּדִים הָאֵלֶּה.
233. אֲנִי מַצִּיעַ פִּתָּרוֹן חָכָם לַבְּעָיָה.
234. הַמְּנַהֵל הִצִּיעַ לִי מִשְׂרָה מְעוּלָּה.
235. הַזַּמָּר מוֹפִיעַ בַּתֵּאַטְרוֹן הַיּוֹם בָּעֶרֶב.
236. הִיא הוֹפִיעָה בַּטֶּלֶוִיזְיָה בַּחֲדָשׁוֹת אֶתְמוֹל.
237. אֲנִי מַבְטִיחַ לַעֲזוֹר לְךָ תָּמִיד.
238. הִבְטַחְתִּי לָבוֹא בַּזְּמַן לַפְּגִישָׁה.
239. עוֹרֵךְ-הַדִּין מוֹכִיחַ אֶת הָאֱמֶת בַּמִּשְׁפָּט.
240. הוֹכַחְתִּי אֶת הַטְּעָנָה שֶׁלִּי בְּבֵרוּר.
241. אֲנַחְנוּ מִתְמוֹדְדִים עִם הַקְּשָׁיִים בְּיַחַד.
242. הִתְמוֹדַדְתִּי הֵיטֵב עִם כָּל הַלַּחַץ.
243. הוּא מִתְנַדֵּב בְּמָדָ״א כָּל שָׁבוּעַ.
244. הִתְנַדַּבְתִּי בַּקְּהִלָּה שָׁנָה שְׁלֵמָה בְּאַהֲבָה.
245. הַיֶּלֶד מִתְנַהֵג יָפֶה מְאֹד בַּכִּתָּה.
246. הִתְנַהַגְתָּ כְּמוֹ בֶּן-אָדָם אֲמִתִּי בַּמַּצָּב.
247. הָעִיר מִתְפַּתַּחַת בִּמְהִירוּת גְּדוֹלָה מְאֹד.
248. הַחֶבְרָה הִתְפַּתְּחָה מְאֹד בַּשָּׁנִים הָאַחֲרוֹנוֹת.
249. אֲנָשִׁים מִתְלוֹנְנִים עַל הַמֶּזֶג-הָאֲוִיר בַּקַּיִץ.
250. הַלָּקוֹחַ הִתְלוֹנֵן עַל הַשֵּׁרוּת הָרַע.
251. הוּא מְטַפֵּל בַּסָּבָא הַיּוֹם.
252. טִפַּלְתִּי בַּחוֹלֶה כָּל הַיּוֹם.
253. הִיא רוֹחֶצֶת אֶת הַיָּדַיִם בְּמַיִם.
254. רָחַצְתִּי אֶת הַפָּנִים בַּבֹּקֶר.
255. הוּא שׁוֹטֵף אֶת הָרִצְפָּה כָּעֵת.
256. שָׁטַפְתִּי כֵּלִים אַחֲרֵי הָאֲרוּחָה.
257. הוּא מַלְבִּישׁ אֶת הַמְּטוּפָּל בִּזְהִירוּת.
258. הִלְבַּשְׁתִּי אֶת הַסָּבָא בְּחוּלְצָה.
259. הִיא מַפְשִׁיטָה אֶת הַמְּעִיל הַכָּבֵד.
260. הִפְשַׁטְתִּי אֶת הַסְּוֶודֶר הַחַם.
261. הוּא מַאֲכִיל אֶת הַסָּבָא בְּמָרָק.
262. הֶאֱכַלְתִּי אֶת הַחוֹלֶה בַּצָּהֳרַיִם.
263. הוּא מַשְׁקֶה אוֹתוֹ בְּמַיִם קָרִים.
264. הִשְׁקֵיתִי אֶת הַסָּבָא בְּתֵה.
265. הוּא מַשְׁכִּיב אֶת הַחוֹלֶה בַּמִּטָּה.
266. הִשְׁכַּבְתִּי אוֹתוֹ לִישׁוֹן בָּעֶרֶב.
267. הוּא מַעֲמִיד אוֹתוֹ עַל הָרַגְלַיִם.
268. הֶעֱמַדְתִּי אֶת הַכִּסֵּא לְיַד הַמִּטָּה.
269. הַמְּטַפֵּל מְסַיֵּעַ לַקָּשִׁישׁ בַּהֲלִיכָה.
270. סִיַּעְתִּי לַמִּשְׁפָּחָה בְּכָל הַסִּדּוּרִים.
271. הוּא תּוֹמֵךְ בּוֹ בִּזְמַן הַהֲלִיכָה.
272. תָּמַכְתִּי בַּסָּבָא עַד הַחֶדֶר.
273. הוּא מְחַמֵּם אֶת הָאוֹכֶל בַּמִּיקְרוֹגַל.

1605. אֲנִי חוֹשֵׁב עָלֶיךָ הַרְבֵּה.
1606. הֵם מְדַבְּרִים עָלָיו עַכְשָׁו.
1607. הוּא עוֹשֶׂה עֲלִיָּה לְיִשְׂרָאֵל.

1611. תַּגִּיד, מָה עוֹבֵר עָלֶיךָ?
1612. הוּא בָּא אֵלַי הַיּוֹם.

1615. הוּא מַגִּיעַ אֵלֶיהָ עַכְשָׁו.

1618. אַל תֵּלֵךְ לְשָׁם בִּלְעָדַי.

1620. אֲנַחְנוּ לֹא נוֹסְעִים בִּלְעָדָיו.

1626. הוּא הוֹלֵךְ אַחֲרֵינוּ עַכְשָׁו.

1712. מָה אַתָּה עוֹשֶׂה הַיּוֹם?
1713. אֲנִי הוֹלֵךְ הַבַּיְתָה עַכְשָׁיו.
1714. שָׁבוּעַ טוֹב לְכוּלָּם!
1715. סְלִיחָה, מָה הַשָּׁעָה עַכְשָׁיו?
1716. יֵשׁ לִי חָבֵר אֶחָד.

1718. הַשָּׁעָה עַכְשָׁיו שָׁלוֹשׁ בְּדִיּוּק.

1724. הַחֲנוּת נִפְתַּחַת בְּשָׁעָה תֵּשַׁע.

1726. זֶה עוֹלֶה עֶשְׂרִים שְׁקָלִים.
1727. הוּא בֶּן שְׁלוֹשִׁים הַיּוֹם.
1728. הַנְּסִיעָה נִמְשֶׁכֶת אַרְבָּעִים דַּקּוֹת.

1730. הַחוּלְצָה עוֹלָה מֵאָה שְׁקָלִים.
1731. הַמַּחְשֵׁב עוֹלֶה אֶלֶף שְׁקָלִים.
